كود وشرح إعداد Cloudflare Worker (الفحص الدقيق 100%)

لماذا نستخدم Cloudflare Worker؟

يفحص هذا الكود اليوزرات عبر واجهة التسجيل الرسمية لإنستغرام (web_create_ajax/attempt) وواجهة البروفايل معاً، وبذلك يفرق بدقة 100% بين اليوزر المتاح للتسجيل فعلياً واليوزر المبند أو المحجوز، ويعطيك 100,000 عملية فحص مجانية يومياً.

خطوات التفعيل في دقيقتين من هاتفك:

  1. افتح موقع dash.cloudflare.com وسجل حساباً مجانياً.
  2. من القائمة الجانبية اختر Workers & Pages ثم اضغط Create application ثم Create Worker.
  3. اضغط Deploy ثم اضغط Edit code وامسح الكود القديم والصق الكود أدناه ثم اضغط Deploy.
  4. انسخ رابط الـ Worker الخاص بك (مثال: https://ig-check.yourname.workers.dev) وضعه في خانة إعدادات الفحص بالموقع.
كود Cloudflare Worker الجاهز (worker.js): 
export default {
  async fetch(request) {
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type"
    };

    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders });
    }

    const url = new URL(request.url);
    const username = (url.searchParams.get("u") || "").trim().toLowerCase();

    if (!username) {
      return new Response(JSON.stringify({ error: "Missing username" }), {
        status: 400,
        headers: { ...corsHeaders, "Content-Type": "application/json" }
      });
    }

    try {
      // Method 1: Official Instagram Web Signup Availability Check
      const csrf = "missing";
      const formBody = new URLSearchParams({
        email: `check_${Math.floor(Math.random() * 999999)}@gmail.com`,
        username: username,
        first_name: "Alex",
        opt_into_one_tap: "false"
      });

      const igSignupRes = await fetch("https://www.instagram.com/api/v1/web/accounts/web_create_ajax/attempt/", {
        method: "POST",
        headers: {
          "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36",
          "X-CSRFToken": csrf,
          "X-IG-App-ID": "936619743392459",
          "X-Requested-With": "XMLHttpRequest",
          "Content-Type": "application/x-www-form-urlencoded",
          "Referer": "https://www.instagram.com/accounts/emailsignup/",
          "Cookie": `csrftoken=${csrf};`
        },
        body: formBody.toString()
      });

      if (igSignupRes.status === 429) {
        return new Response(JSON.stringify({
          username,
          status: "rate_limited",
          detail: "Rate limited by Instagram"
        }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
      }

      if (igSignupRes.ok) {
        const data = await igSignupRes.json();
        if (data && data.errors) {
          const hasUsernameError = Array.isArray(data.errors.username) && data.errors.username.length > 0;
          if (!hasUsernameError && data.account_created === false) {
            return new Response(JSON.stringify({
              username,
              status: "available",
              method: "signup_api",
              detail: "متاح للتسجيل رسمياً 100%"
            }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
          } else if (hasUsernameError) {
            const errCode = data.errors.username[0]?.code || "username_is_taken";
            return new Response(JSON.stringify({
              username,
              status: "taken",
              method: "signup_api",
              detail: errCode
            }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
          }
        }
      }

      // Fallback Method 2: Web Profile Info Endpoint
      const profileRes = await fetch(`https://www.instagram.com/api/v1/users/web_profile_info/?username=${encodeURIComponent(username)}`, {
        headers: {
          "User-Agent": "Mozilla/5.0 (Linux; Android 13; SM-S908B) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Mobile Safari/537.36",
          "X-IG-App-ID": "936619743392459"
        }
      });

      if (profileRes.status === 404) {
        return new Response(JSON.stringify({
          username,
          status: "available",
          method: "profile_api",
          detail: "غير مستخدم (404)"
        }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
      } else if (profileRes.status === 200) {
        return new Response(JSON.stringify({
          username,
          status: "taken",
          method: "profile_api",
          detail: "مستخدم حالياً"
        }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
      }

      return new Response(JSON.stringify({
        username,
        status: "unknown",
        detail: `HTTP ${profileRes.status}`
      }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });

    } catch (err) {
      return new Response(JSON.stringify({
        username,
        status: "error",
        detail: err.message
      }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } });
    }
  }
};
